// Service d'alertes Disneyland Paris (Cloudflare Worker) // Vérifie les temps d'attente chaque minute et envoie une notification via ntfy.sh const DEST = 'e8d0207f-da8a-4048-bec8-117aa946b2c2'; const API = 'https://api.themeparks.wiki/v1/entity/' + DEST + '/live'; const ORIGIN = 'https://jul45000.github.io'; const PAGE = ORIGIN + '/disney-live/'; const TTL = 3 * 24 * 3600; // les alertes expirent après 3 jours sans synchronisation const REFRESH = 12 * 3600 * 1000; // rafraîchit l'expiration au plus toutes les 12 h const cors = { 'Access-Control-Allow-Origin': ORIGIN, 'Access-Control-Allow-Methods': 'POST, GET, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Vary': 'Origin' }; const json = (o, s = 200) => new Response(JSON.stringify(o), { status: s, headers: Object.assign({ 'Content-Type': 'application/json' }, cors) }); const okTopic = t => typeof t === 'string' && /^[A-Za-z0-9_-]{16,64}$/.test(t); const waitOf = a => a.status === 'OPERATING' && a.queue && a.queue.STANDBY && a.queue.STANDBY.waitTime != null ? a.queue.STANDBY.waitTime : null; async function push(topic, title, message) { const r = await fetch('https://ntfy.sh/' + topic, { method: 'POST', headers: { Title: title, Priority: '5', Tags: 'tada', Click: PAGE }, body: message }); return r; } async function run(env) { const list = await env.ALERTS.list({ prefix: 'u:' }); if (!list.keys.length) return; const r = await fetch(API, { headers: { 'User-Agent': 'dlp-live-alerts' } }); if (!r.ok) return; const j = await r.json(); const byId = {}; (j.liveData || []).forEach(a => { byId[a.id] = a; }); for (const k of list.keys) { const rec = await env.ALERTS.get(k.name, 'json'); if (!rec || !rec.alerts) continue; const topic = k.name.slice(2); let changed = false; for (const id of Object.keys(rec.alerts)) { const al = rec.alerts[id], a = byId[id]; if (!a) continue; const w = waitOf(a); if (w == null) continue; if (w <= al.t) { if (!al.fired) { al.fired = true; changed = true; try { const pr = await push(topic, 'Attente atteinte', al.name + ' : ' + w + ' min (cible ' + al.t + ' min ou moins)'); if (!pr.ok) { al.fired = false; } } catch (e) { al.fired = false; } } } else if (al.fired) { al.fired = false; changed = true; } } if (changed) await env.ALERTS.put(k.name, JSON.stringify(rec), { expirationTtl: TTL }); } } export default { async fetch(req, env) { if (req.method === 'OPTIONS') return new Response(null, { headers: cors }); if (req.method === 'GET') return json({ ok: true, service: 'dlp-alerts' }); const url = new URL(req.url); let b; try { b = await req.json(); } catch (e) { return json({ error: 'json' }, 400); } if (!b || !okTopic(b.topic)) return json({ error: 'topic' }, 400); const key = 'u:' + b.topic; if (url.pathname === '/test') { const pr = await push(b.topic, 'Test Disneyland Live', 'Les notifications fonctionnent !'); if (!pr.ok) return json({ error: 'ntfy', status: pr.status }, pr.status >= 400 ? pr.status : 502); return json({ ok: true }); } if (url.pathname === '/sync') { const inc = b.alerts && typeof b.alerts === 'object' ? b.alerts : {}; const old = await env.ALERTS.get(key, 'json'); const oldA = old && old.alerts ? old.alerts : {}; const out = {}; let n = 0; for (const id of Object.keys(inc)) { if (++n > 50) break; const t = Number(inc[id] && inc[id].t); if (!Number.isFinite(t) || t < 0 || t > 240) continue; out[id] = { t: t, name: String((inc[id] && inc[id].name) || '').slice(0, 80), fired: !!(oldA[id] && oldA[id].t === t && oldA[id].fired) }; } const count = Object.keys(out).length; if (!count) { if (old) await env.ALERTS.delete(key); return json({ ok: true, count: 0 }); } const same = old && JSON.stringify(oldA) === JSON.stringify(out) && Date.now() - (old.ts || 0) < REFRESH; if (!same) await env.ALERTS.put(key, JSON.stringify({ ts: Date.now(), alerts: out }), { expirationTtl: TTL }); return json({ ok: true, count: count }); } return json({ error: 'route' }, 404); }, async scheduled(event, env, ctx) { ctx.waitUntil(run(env)); } };
